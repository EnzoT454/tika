package org.apache.tika.io;

import org.mockito.*;
import org.junit.jupiter.api.*;

@ExtendWith(MockitoExtension.class)
public class FilenameUtils_getSanitizedEmbeddedFileName_3_1_Test {

    @InjectMocks
    private FilenameUtils filenameUtils;

    @Mock
    private Metadata metadata;

    @Test
    public void testGetSanitizedEmbeddedFileName_withValidPath() throws IOException {
        when(metadata.get(FilenameUtils.TikaCoreProperties.RESOURCE_NAME_KEY)).thenReturn("example.txt");
        String result = filenameUtils.getSanitizedEmbeddedFileName(metadata, "txt", 10);
        assertEquals("example.txt", result);
    }

    @Test
    public void testGetSanitizedEmbeddedFileName_withBlankPath() throws IOException {
        when(metadata.get(FilenameUtils.TikaCoreProperties.RESOURCE_NAME_KEY)).thenReturn("");
        String result = filenameUtils.getSanitizedEmbeddedFileName(metadata, "txt", 10);
        assertEquals(null, result);
    }

    @Test
    public void testGetSanitizedEmbeddedFileName_withNullPath() throws IOException {
        when(metadata.get(FilenameUtils.TikaCoreProperties.RESOURCE_NAME_KEY)).thenReturn(null);
        String result = filenameUtils.getSanitizedEmbeddedFileName(metadata, "txt", 10);
        assertEquals(null, result);
    }

    @Test
    public void testGetSanitizedEmbeddedFileName_withProtocolPath() throws IOException {
        when(metadata.get(FilenameUtils.TikaCoreProperties.RESOURCE_NAME_KEY)).thenReturn("http://example.com/example.txt");
        String result = filenameUtils.getSanitizedEmbeddedFileName(metadata, "txt", 10);
        assertEquals("example.txt", result);
    }

    @Test
    public void testGetSanitizedEmbeddedFileName_withReservedCharacters() throws IOException {
        when(metadata.get(FilenameUtils.TikaCoreProperties.RESOURCE_NAME_KEY)).thenReturn("example?file.txt");
        String result = filenameUtils.getSanitizedEmbeddedFileName(metadata, "txt", 10);
        assertEquals("example%3Ffile.txt", result);
    }

    @Test
    public void testGetSanitizedEmbeddedFileName_withLongFileName() throws IOException {
        when(metadata.get(FilenameUtils.TikaCoreProperties.RESOURCE_NAME_KEY)).thenReturn("a".repeat(100) + ".txt");
        String result = filenameUtils.getSanitizedEmbeddedFileName(metadata, "txt", 10);
        assertEquals("a".repeat(9) + "....txt", result);
    }

    @Test
    public void testGetSanitizedEmbeddedFileName_withEmptyExtension() throws IOException {
        when(metadata.get(FilenameUtils.TikaCoreProperties.RESOURCE_NAME_KEY)).thenReturn("example.");
        String result = filenameUtils.getSanitizedEmbeddedFileName(metadata, "txt", 10);
        assertEquals(null, result);
    }

    @Test
    public void testGetSanitizedEmbeddedFileName_withASCIINumericExtension() throws IOException {
        when(metadata.get(FilenameUtils.TikaCoreProperties.RESOURCE_NAME_KEY)).thenReturn("example.12345");
        String result = filenameUtils.getSanitizedEmbeddedFileName(metadata, "txt", 10);
        assertEquals("example.12345", result);
    }

    @Test
    public void testGetSanitizedEmbeddedFileName_withInvalidASCIINumericExtension() throws IOException {
        when(metadata.get(FilenameUtils.TikaCoreProperties.RESOURCE_NAME_KEY)).thenReturn("example.123456");
        String result = filenameUtils.getSanitizedEmbeddedFileName(metadata, "txt", 10);
        assertEquals("example.txt", result);
    }

    @Test
    public void testGetSanitizedEmbeddedFileName_withMultipleDots() throws IOException {
        when(metadata.get(FilenameUtils.TikaCoreProperties.RESOURCE_NAME_KEY)).thenReturn("example..txt");
        String result = filenameUtils.getSanitizedEmbeddedFileName(metadata, "txt", 10);
        assertEquals("example.txt", result);
    }

    @Test
    public void testGetSanitizedEmbeddedFileName_withLeadingDot() throws IOException {
        when(metadata.get(FilenameUtils.TikaCoreProperties.RESOURCE_NAME_KEY)).thenReturn(".example.txt");
        String result = filenameUtils.getSanitizedEmbeddedFileName(metadata, "txt", 10);
        assertEquals("_example.txt", result);
    }

    @Test
    public void testGetSanitizedEmbeddedFileName_withTrailingDot() throws IOException {
        when(metadata.get(FilenameUtils.TikaCoreProperties.RESOURCE_NAME_KEY)).thenReturn("example.txt.");
        String result = filenameUtils.getSanitizedEmbeddedFileName(metadata, "txt", 10);
        assertEquals("example.txt", result);
    }
}
