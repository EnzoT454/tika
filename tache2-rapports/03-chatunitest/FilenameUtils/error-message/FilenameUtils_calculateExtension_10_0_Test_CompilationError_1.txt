    @Test
    public void testCalculateExtension_withMimeType() throws MimeTypeException {
        String mimeType = "image/png";
        when(mimeTypes.forName(mimeType)).thenReturn(mock(MimeType.class));
        when(mimeTypes.forName(mimeType).getExtension()).thenReturn("png");
        Metadata metadata = new Metadata();
        metadata.set(Metadata.CONTENT_TYPE, mimeType);
        String result = filenameUtils.calculateExtension(metadata, "default");
        assertEquals("png", result);
    }
