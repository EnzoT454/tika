package org.apache.tika.io;

import org.apache.tika.metadata.Metadata;
import org.apache.tika.utils.StringUtils;
import org.junit.jupiter.api.extension.ExtendWith;
import org.mockito.junit.jupiter.MockitoExtension;
import java.io.IOException;
import org.mockito.*;
import org.junit.jupiter.api.*;
import static org.mockito.Mockito.*;
import static org.junit.jupiter.api.Assertions.*;
import java.nio.file.Path;
import java.util.HashSet;
import java.util.Locale;

@ExtendWith(MockitoExtension.class)
public class FilenameUtils_getSanitizedEmbeddedFileName_3_0_Test {

    @Mock
    private Metadata metadata;

    private FilenameUtils filenameUtils;

    @BeforeEach
    public void setUp() {
        filenameUtils = new FilenameUtils();
    }

    @Test
    public void testGetSanitizedEmbeddedFileName_withValidPath() throws IOException {
        when(metadata.get(anyString())).thenReturn("example.txt");
        String result = filenameUtils.getSanitizedEmbeddedFileName(metadata, "txt", 10);
        assertEquals("example.txt", result);
    }

    @Test
    public void testGetSanitizedEmbeddedFileName_withNullPath() throws IOException {
        when(metadata.get(anyString())).thenReturn(null);
        String result = filenameUtils.getSanitizedEmbeddedFileName(metadata, "txt", 10);
        assertNull(result);
    }

    @Test
    public void testGetSanitizedEmbeddedFileName_withEmptyPath() throws IOException {
        when(metadata.get(anyString())).thenReturn("");
        String result = filenameUtils.getSanitizedEmbeddedFileName(metadata, "txt", 10);
        assertNull(result);
    }

    @Test
    public void testGetSanitizedEmbeddedFileName_withInvalidPath() throws IOException {
        when(metadata.get(anyString())).thenReturn("\"example.txt\"");
        String result = filenameUtils.getSanitizedEmbeddedFileName(metadata, "txt", 10);
        assertEquals("example.txt", result);
    }

    @Test
    public void testGetSanitizedEmbeddedFileName_withLongPath() throws IOException {
        when(metadata.get(anyString())).thenReturn("a".repeat(100));
        String result = filenameUtils.getSanitizedEmbeddedFileName(metadata, "txt", 10);
        assertEquals("a".repeat(10) + "....txt", result);
    }

    @Test
    public void testGetSanitizedEmbeddedFileName_withExtension() throws IOException {
        when(metadata.get(anyString())).thenReturn("example.docx");
        String result = filenameUtils.getSanitizedEmbeddedFileName(metadata, "txt", 10);
        assertEquals("example.docx", result);
    }

    @Test
    public void testGetSanitizedEmbeddedFileName_withEmptyExtension() throws IOException {
        when(metadata.get(anyString())).thenReturn("example.");
        String result = filenameUtils.getSanitizedEmbeddedFileName(metadata, "txt", 10);
        assertNull(result);
    }

    @Test
    public void testGetSanitizedEmbeddedFileName_withNullExtension() throws IOException {
        when(metadata.get(anyString())).thenReturn("example");
        String result = filenameUtils.getSanitizedEmbeddedFileName(metadata, "txt", 10);
        assertEquals("example.txt", result);
    }

    @Test
    public void testGetSanitizedEmbeddedFileName_withInvalidExtension() throws IOException {
        when(metadata.get(anyString())).thenReturn("example..txt");
        String result = filenameUtils.getSanitizedEmbeddedFileName(metadata, "txt", 10);
        assertEquals("example..txt", result);
    }

    @Test
    public void testGetSanitizedEmbeddedFileName_withLeadingDot() throws IOException {
        when(metadata.get(anyString())).thenReturn(".example.txt");
        String result = filenameUtils.getSanitizedEmbeddedFileName(metadata, "txt", 10);
        assertEquals("_example.txt", result);
    }

    @Test
    public void testGetSanitizedEmbeddedFileName_withMultipleDots() throws IOException {
        when(metadata.get(anyString())).thenReturn("example..txt");
        String result = filenameUtils.getSanitizedEmbeddedFileName(metadata, "txt", 10);
        assertEquals("example..txt", result);
    }

    @Test
    public void testGetSanitizedEmbeddedFileName_withInvalidCharacters() throws IOException {
        when(metadata.get(anyString())).thenReturn("example*txt");
        String result = filenameUtils.getSanitizedEmbeddedFileName(metadata, "txt", 10);
        assertEquals("example%2A.txt", result);
    }

    @Test
    public void testGetSanitizedEmbeddedFileName_withReservedCharacters() throws IOException {
        when(metadata.get(anyString())).thenReturn("example<txt");
        String result = filenameUtils.getSanitizedEmbeddedFileName(metadata, "txt", 10);
        assertEquals("example%3C.txt", result);
    }

    @Test
    public void testGetSanitizedEmbeddedFileName_withEmptyNamePart() throws IOException {
        when(metadata.get(anyString())).thenReturn(".txt");
        String result = filenameUtils.getSanitizedEmbeddedFileName(metadata, "txt", 10);
        assertNull(result);
    }
}
