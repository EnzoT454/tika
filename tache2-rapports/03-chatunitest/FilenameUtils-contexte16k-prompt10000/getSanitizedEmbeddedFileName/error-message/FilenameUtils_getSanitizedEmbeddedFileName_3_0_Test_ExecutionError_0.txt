@ExtendWith(MockitoExtension.class)
public class FilenameUtils_getSanitizedEmbeddedFileName_3_0_Test {

    @Mock
    private Metadata metadata;

    @BeforeEach
    public void setUp() {
        when(metadata.get(eq(TikaCoreProperties.RESOURCE_NAME_KEY))).thenReturn("example.txt");
    }

    @Test
    public void testGetSanitizedEmbeddedFileName_withValidInput() throws Exception {
        FilenameUtils filenameUtils = new FilenameUtils();
        String result = filenameUtils.getSanitizedEmbeddedFileName(metadata, "txt", 10);
        assertEquals("example.txt", result);
    }

    @Test
    public void testGetSanitizedEmbeddedFileName_withNullInput() throws Exception {
        FilenameUtils filenameUtils = new FilenameUtils();
        when(metadata.get(eq(TikaCoreProperties.RESOURCE_NAME_KEY))).thenReturn(null);
        String result = filenameUtils.getSanitizedEmbeddedFileName(metadata, "txt", 10);
        assertEquals(null, result);
    }

    @Test
    public void testGetSanitizedEmbeddedFileName_withEmptyString() throws Exception {
        FilenameUtils filenameUtils = new FilenameUtils();
        when(metadata.get(eq(TikaCoreProperties.RESOURCE_NAME_KEY))).thenReturn("");
        String result = filenameUtils.getSanitizedEmbeddedFileName(metadata, "txt", 10);
        assertEquals(null, result);
    }

    @Test
    public void testGetSanitizedEmbeddedFileName_withPath() throws Exception {
        FilenameUtils filenameUtils = new FilenameUtils();
        when(metadata.get(eq(TikaCoreProperties.RESOURCE_NAME_KEY))).thenReturn("/path/to/example.txt");
        String result = filenameUtils.getSanitizedEmbeddedFileName(metadata, "txt", 10);
        assertEquals("example.txt", result);
    }

    @Test
    public void testGetSanitizedEmbeddedFileName_withProtocol() throws Exception {
        FilenameUtils filenameUtils = new FilenameUtils();
        when(metadata.get(eq(TikaCoreProperties.RESOURCE_NAME_KEY))).thenReturn("http://example.com/example.txt");
        String result = filenameUtils.getSanitizedEmbeddedFileName(metadata, "txt", 10);
        assertEquals("example.txt", result);
    }

    @Test
    public void testGetSanitizedEmbeddedFileName_withExtension() throws Exception {
        FilenameUtils filenameUtils = new FilenameUtils();
        when(metadata.get(eq(TikaCoreProperties.RESOURCE_NAME_KEY))).thenReturn("example.docx");
        String result = filenameUtils.getSanitizedEmbeddedFileName(metadata, "txt", 10);
        assertEquals("example.txt", result);
    }

    @Test
    public void testGetSanitizedEmbeddedFileName_withBlankExtension() throws Exception {
        FilenameUtils filenameUtils = new FilenameUtils();
        when(metadata.get(eq(TikaCoreProperties.RESOURCE_NAME_KEY))).thenReturn("example.");
        String result = filenameUtils.getSanitizedEmbeddedFileName(metadata, "txt", 10);
        assertEquals(null, result);
    }

    @Test
    public void testGetSanitizedEmbeddedFileName_withTooLongName() throws Exception {
        FilenameUtils filenameUtils = new FilenameUtils();
        when(metadata.get(eq(TikaCoreProperties.RESOURCE_NAME_KEY))).thenReturn("a".repeat(100));
        String result = filenameUtils.getSanitizedEmbeddedFileName(metadata, "txt", 10);
        assertEquals("a".repeat(7) + "...txt", result);
    }

    @Test
    public void testGetSanitizedEmbeddedFileName_withReservedCharacters() throws Exception {
        FilenameUtils filenameUtils = new FilenameUtils();
        when(metadata.get(eq(TikaCoreProperties.RESOURCE_NAME_KEY))).thenReturn("example?*<>|\":/");
        String result = filenameUtils.getSanitizedEmbeddedFileName(metadata, "txt", 10);
        assertEquals("example_%25_2A_3C_3E_7C_22_3A_2F.txt", result);
    }
}
