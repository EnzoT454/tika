package org.apache.tika.io;

import java.io.ByteArrayInputStream;
import java.io.IOException;
import org.mockito.*;
import org.junit.jupiter.api.*;
import static org.mockito.Mockito.*;
import static org.junit.jupiter.api.Assertions.*;
import org.junit.jupiter.api.extension.ExtendWith;
import org.mockito.junit.jupiter.MockitoExtension;
import java.io.InputStream;

public class LookaheadInputStream_skip_4_0_Test {

    private LookaheadInputStream lookaheadInputStream;

    @BeforeEach
    public void setUp() throws IOException {
        // Create a ByteArrayInputStream with some data
        byte[] data = { 1, 2, 3, 4, 5 };
        ByteArrayInputStream inputStream = new ByteArrayInputStream(data);
        // Initialize the LookaheadInputStream with the input stream and buffer size
        lookaheadInputStream = new LookaheadInputStream(inputStream, 5);
    }

    @Test
    public void testSkip() throws IOException {
        // Skip 2 bytes
        long skipped = lookaheadInputStream.skip(2);
        // Check that the position has been updated correctly
        assertEquals(2, skipped);
        assertEquals(2, lookaheadInputStream.position);
        // Check that the available bytes have been reduced
        assertEquals(3, lookaheadInputStream.available());
    }

    @Test
    public void testSkipBeyondAvailable() throws IOException {
        // Skip 6 bytes (more than available)
        long skipped = lookaheadInputStream.skip(6);
        // Check that the position has been updated correctly
        assertEquals(5, skipped);
        assertEquals(5, lookaheadInputStream.position);
        // Check that the available bytes have been reduced to 0
        assertEquals(0, lookaheadInputStream.available());
    }
}
